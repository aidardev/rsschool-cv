Code Example

function throttle<T extends (...args: any[]) => any>(fn: T, delay: number = 300) {
    let lastCall = 0;
    let lastResult: ReturnType<T>;

    return function (this: ThisParameterType<T>, ...args: Parameters<T>) {
        const now = Date.now();

        if (now - lastCall >= delay) {
            lastCall = now;
            lastResult = fn.apply(this, args);
        }

        return lastResult;
    };
}
